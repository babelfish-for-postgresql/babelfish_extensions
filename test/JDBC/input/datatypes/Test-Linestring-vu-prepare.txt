CREATE VIEW DimOfGeometryline AS SELECT location.STAsText() AS geometry, location.STDimension() AS dimension FROM GEOSPATIALLINEGEOM_dt ORDER BY id;

CREATE VIEW AreaOfGeometryline AS SELECT location.STAsText() AS geometry, location.STArea() AS area FROM GEOSPATIALLINEGEOM_dt ORDER BY id;

CREATE VIEW TextFromGeometryline AS SELECT location.STAsText() AS text FROM GEOSPATIALLINEGEOM_dt ORDER BY id;

CREATE VIEW BinaryFromGeometryline AS SELECT location.STAsText() AS geometry, location.STAsBinary() AS binary FROM GEOSPATIALLINEGEOM_dt ORDER BY id;

CREATE VIEW SridFromGeometryline AS SELECT location.STAsText() AS geometry, location.STSrid AS srid FROM GEOSPATIALLINEGEOM_dt ORDER BY id;

CREATE VIEW EmptyGeometryline AS SELECT location.STAsText() AS geometry, location.STIsEmpty() AS isempty FROM GEOSPATIALLINEGEOM_dt ORDER BY id;

CREATE VIEW ValidGeometryline AS SELECT location.STAsText() AS geometry, location.STIsValid() AS isvalid FROM GEOSPATIALLINEGEOM_dt ORDER BY id;

CREATE VIEW ClosedGeometryline AS SELECT location.STAsText() AS geometry, location.STIsClosed() AS isclosed FROM GEOSPATIALLINEGEOM_dt ORDER BY id;

CREATE VIEW DisjointTempGeomline AS SELECT p1.location.STAsText() AS geometry1, p2.location.STAsText() AS geometry2, p1.location.STDisjoint(p2.location) AS disjoints FROM GEOSPATIALLINEGEOM_dt p1 CROSS JOIN GEOSPATIALLINEGEOM_dt p2 ORDER BY p1.id;

CREATE VIEW DistanceTempGeomline AS SELECT p1.location.STAsText() AS geometry1, p2.location.STAsText() AS geometry2,  CAST(STDistance( p1.location, p2.location ) AS numeric(20, 6)) AS distance FROM GEOSPATIALLINEGEOM_dt p1 CROSS JOIN GEOSPATIALLINEGEOM_dt p2 ORDER BY p1.id;

CREATE VIEW ContainTempGeomline AS SELECT p1.location.STAsText() AS geometry1, p2.location.STAsText() AS geometry2, p1.location.STContains(p2.location) AS contain FROM GEOSPATIALLINEGEOM_dt p1 CROSS JOIN GEOSPATIALLINEGEOM_dt p2 ORDER BY p1.id;
