# SqlBulkCopy to Temp Table Tests
# Tests the sp_tablecollations_100 implementation for BCP to temp tables

# SECTION 1: SqlBulkCopy to ENR Temp Tables (Standard temp tables stored in currentQueryEnv)

# Test 1: Basic ENR Temp Table
Create table #EnrTempTable1(id int, name varchar(50))
sqlbulkcopy_temptable#!##EnrTempTable1#!#5
Select * from #EnrTempTable1 order by id
drop table #EnrTempTable1

# Test 2: ENR Temp Table with more rows
Create table #EnrTempTable2(id int, name varchar(50))
sqlbulkcopy_temptable#!##EnrTempTable2#!#10
Select count(*) as row_count from #EnrTempTable2
drop table #EnrTempTable2

# Test 3: ENR Temp Table - verify data integrity
Create table #EnrTempTable3(id int, name varchar(50))
sqlbulkcopy_temptable#!##EnrTempTable3#!#3
Select id, name from #EnrTempTable3 order by id
drop table #EnrTempTable3

# SECTION 2: SqlBulkCopy to Non-ENR Temp Tables (Temp tables using UDT, stored in pg_temp schema)

# Test 4: Non-ENR Temp Table (using UDT)

sqlbulkcopy_temptable#!##NonEnrTempTable4#!#5

drop table #NonEnrTempTable4

# Test 5: Non-ENR Temp Table with more rows
BEGIN TRY CREATE TYPE dbo.BcpTestVarcharType2 FROM VARCHAR(100) END TRY BEGIN CATCH END CATCH

Select count(*) as row_count from #NonEnrTempTable5
drop table #NonEnrTempTable5
